cmake_minimum_required(VERSION 3.24)
project(GenForm VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network Concurrent OpenGL Svg PrintSupport)
set(GENFORM_OCCT_ROOT "" CACHE PATH "OpenCascade SDK prefix (include/opencascade and lib)")
if(NOT GENFORM_OCCT_ROOT)
    find_package(OpenCASCADE QUIET COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
endif()
if(NOT OpenCASCADE_FOUND OR GENFORM_OCCT_ROOT)
    find_path(GENFORM_OCCT_INCLUDE_DIR Standard.hxx HINTS "${GENFORM_OCCT_ROOT}" PATH_SUFFIXES include/opencascade opencascade)
    if(GENFORM_OCCT_INCLUDE_DIR)
        set(OpenCASCADE_LIBRARIES)
        foreach(component TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKFillet TKOffset TKFeat TKShHealing TKMesh TKHLR TKXSBase TKV3d TKService)
            find_library(GENFORM_OCCT_${component} NAMES ${component} HINTS "${GENFORM_OCCT_ROOT}/lib" REQUIRED)
            list(APPEND OpenCASCADE_LIBRARIES "${GENFORM_OCCT_${component}}")
        endforeach()
        find_library(GENFORM_OCCT_STEP NAMES TKDESTEP TKSTEP HINTS "${GENFORM_OCCT_ROOT}/lib" REQUIRED)
        find_library(GENFORM_OCCT_STL NAMES TKDESTL TKSTL HINTS "${GENFORM_OCCT_ROOT}/lib" REQUIRED)
        list(APPEND OpenCASCADE_LIBRARIES "${GENFORM_OCCT_STEP}" "${GENFORM_OCCT_STL}")
        set(OpenCASCADE_INCLUDE_DIR "${GENFORM_OCCT_INCLUDE_DIR}")
        set(OpenCASCADE_FOUND TRUE)
    endif()
endif()

file(GLOB_RECURSE GENFORM_SOURCES CONFIGURE_DEPENDS genform/*.cpp genform/*.hpp)
list(FILTER GENFORM_SOURCES EXCLUDE REGEX "/\\.native-build[^/]*/")
list(FILTER GENFORM_SOURCES EXCLUDE REGEX "/native_selftest\\.cpp$")
add_library(genform_core STATIC ${GENFORM_SOURCES})
target_include_directories(genform_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(genform_core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent Qt6::OpenGL Qt6::Svg Qt6::PrintSupport)
target_compile_definitions(genform_core PUBLIC GENFORM_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
    target_link_libraries(genform_core PUBLIC dwmapi user32)
endif()
if(OpenCASCADE_FOUND)
    target_include_directories(genform_core PUBLIC ${OpenCASCADE_INCLUDE_DIR})
    target_link_directories(genform_core PUBLIC ${OpenCASCADE_LIBRARY_DIR})
    target_link_libraries(genform_core PUBLIC ${OpenCASCADE_LIBRARIES})
    target_compile_definitions(genform_core PUBLIC GENFORM_HAS_OCCT=1)
else()
    message(WARNING "OpenCascade C++ SDK not found. Geometry commands report an explicit dependency error; document/API tools remain available.")
endif()
if(MSVC)
    target_compile_options(genform_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
else()
    target_compile_options(genform_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(genform main.cpp)
target_link_libraries(genform PRIVATE genform_core)
add_executable(genform_mcp genform_mcp.cpp)
target_link_libraries(genform_mcp PRIVATE genform_core)

set_property(TARGET genform genform_mcp PROPERTY RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" config_upper)
    set_property(TARGET genform genform_mcp PROPERTY RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}")
endforeach()

file(GLOB GENFORM_QML_RESOURCES CONFIGURE_DEPENDS qml/*.qml qml/qmldir)
file(GLOB GENFORM_ASSET_RESOURCES CONFIGURE_DEPENDS genform/assets/*)
qt_add_resources(genform_core command_catalog PREFIX "/genform" BASE "${CMAKE_CURRENT_SOURCE_DIR}/genform" FILES genform/commands.json)
qt_add_resources(genform_core qml_resources PREFIX "/genform" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${GENFORM_QML_RESOURCES})
qt_add_resources(genform_core asset_resources PREFIX "/genform" BASE "${CMAKE_CURRENT_SOURCE_DIR}/genform" FILES ${GENFORM_ASSET_RESOURCES})

    add_test(NAME gui_smoke_test COMMAND genform --gui-smoke-test)
    set_tests_properties(gui_smoke_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;GENFORM_PORT=0" TIMEOUT 180)

option(GENFORM_BUILD_EXAMPLES "Build native modeling and verification examples" ON)
if(GENFORM_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()

install(TARGETS genform genform_mcp RUNTIME DESTINATION bin)
install(DIRECTORY qml genform/assets DESTINATION share/genform)
install(FILES genform/commands.json DESTINATION share/genform)
