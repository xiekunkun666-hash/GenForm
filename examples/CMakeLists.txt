cmake_minimum_required(VERSION 3.24)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GenFormExamples LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_AUTOMOC ON)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2 Network Concurrent OpenGL Svg PrintSupport)
    find_path(GENFORM_OCCT_INCLUDE_DIR Standard.hxx PATH_SUFFIXES include/opencascade opencascade REQUIRED)
    set(GENFORM_OCCT_LIBRARIES)
    foreach(component TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool TKFillet TKOffset TKFeat TKShHealing TKMesh TKHLR TKXSBase TKDESTEP TKDESTL TKV3d TKService)
        find_library(GENFORM_OCCT_${component} NAMES ${component} REQUIRED)
        list(APPEND GENFORM_OCCT_LIBRARIES "${GENFORM_OCCT_${component}}")
    endforeach()
    get_filename_component(GENFORM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
    file(GLOB_RECURSE GENFORM_EXAMPLE_CORE_SOURCES CONFIGURE_DEPENDS "${GENFORM_ROOT}/genform/*.cpp" "${GENFORM_ROOT}/genform/*.hpp")
    list(FILTER GENFORM_EXAMPLE_CORE_SOURCES EXCLUDE REGEX "/[^/]*selftest\\.cpp$")
    add_library(genform_core STATIC ${GENFORM_EXAMPLE_CORE_SOURCES})
    target_include_directories(genform_core PUBLIC "${GENFORM_ROOT}")
    target_compile_definitions(genform_core PUBLIC GENFORM_SOURCE_DIR="${GENFORM_ROOT}")
    target_link_libraries(genform_core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Concurrent Qt6::OpenGL Qt6::Svg Qt6::PrintSupport)
    target_include_directories(genform_core PUBLIC "${GENFORM_OCCT_INCLUDE_DIR}")
    target_link_libraries(genform_core PUBLIC ${GENFORM_OCCT_LIBRARIES})
    target_compile_definitions(genform_core PUBLIC GENFORM_HAS_OCCT=1)
    if(MSVC)
        target_compile_options(genform_core PUBLIC /utf-8 /W4 /permissive-)
    endif()
endif()

add_executable(genform_examples native/main.cpp native/runner.cpp native/verification.cpp native/surfaces.cpp native/presentation.cpp native/skye.cpp native/alternate.cpp)
target_link_libraries(genform_examples PRIVATE genform_core)
target_compile_definitions(genform_examples PRIVATE GENFORM_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(genform_examples_kernel_audit native/kernel_audit.cpp)
target_link_libraries(genform_examples_kernel_audit PRIVATE genform_core)
target_compile_definitions(genform_examples_kernel_audit PRIVATE GENFORM_EXAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(genform_examples PRIVATE /utf-8 /W4 /permissive-)
else()
    target_compile_options(genform_examples PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_test(NAME examples_recipe_audit COMMAND genform_examples --audit)
add_test(NAME examples_native_self_test COMMAND genform_examples --self-test)
add_test(NAME examples_comparison_self_test COMMAND genform_examples porsche_911/compare --self-test)
add_test(NAME examples_journal_self_test COMMAND genform_examples porsche_911/alternate_runtime)
add_test(NAME examples_kernel_recipe_audit COMMAND genform_examples_kernel_audit all "${CMAKE_CURRENT_BINARY_DIR}/verification/all-recipes-kernel.json")
set_tests_properties(examples_kernel_recipe_audit PROPERTIES TIMEOUT 180)
